1. Context Switch
    - 커널 진입 경로
        
        (1) Interrupt: HW가 CPU에게 보내는 비동기 신호→ 예측X
        
        (2) Trap: ‘’ 동기적 신호 ex. system call, 예외
        
        → 이걸로 커널 진입(mode switch) → 그 때 다른 상태가 되면(context swtich)
        
        |  | 커널 진입 방법 |
        | --- | --- |
        | IO 시작 | Trap |
        | ‘ ‘ 완료 | Interrupt |
        | 타이머 | Interrupt |
        | 비블로킹콜(전이 X) | Trap |
    - 상태 전이 예시
        - waiting → ready: 상태가 ready인 것을 ready queue에
        - running → running: getpid()같이 하고 바로 다시 실행할 수 있는 거
            - running → waiting: read()
    - Context Switch
        
        Context: 프로세스 재개를 위해 필요한 CPU 상태 전체
        
        현재 프로세스 상태를 PCB에 저장하고, 다음 프로세스의 상태를 PCB에 복원해서 실행을 전환하는 것
        
    - Trap에서 저장하는 순서
        
        (1) HW(CPU)가 저장
        
        - 커널 스택에
        - 저장 regs
            
            ① 스택 포인터: SS, RSP
            
            ② 플래그 레지스터: RFLAGS
            
            ③ 복귀 주소: CS, RIP
            
        
        (2) SW(커널)가 추가 저장
        
        - 커널 스택에
        - 저장 레지스터: 범용, ..
2. fork()의 동작 원리
    
    (1) 자식이 부모와 같은 주소 공간으로 시작(CoW: Copy on Write, 물리메모리를 공유하다가 쓰기 시점에 복사 ⇒ 자식에서 값 바꿔도 다 안 바뀜 global이든 local이든)
    
    - code
        
        ```c
        #include <stdio.h>
        #include <unistd.h>
        
        int global = 10;
        
        int main(void) {
            int local = 20;
            pid_t pid = fork();
        
            if (pid == 0) {
                // 자식: 값변경
                global = 99;
                local = 88;
                printf("child: g=%d, l=%d\n", global , local);
            } else {
                sleep (1); // 자식먼저실행
                printf("parent: g=%d, l=%d\n", global , local);
            }
            return 0;
        }
        ```
        
    
    (2) 커널이 자식용 PCB 생성
    
    (3) 자식에게 새 PID 할당
    
    (4) fd 테이블은 복사, open file entry랑 파일 오프셋은 공유
    
    - fd: 프로세스가 열린 파일, 장치를 가리키는 정수 번호
    - 0: stdin, 1: stdout, 2: stderr
    
    (5) 부모, 자식 모두 fork() 다음 줄 재개
    
    - 반환값
        - 부모: 자식 PID
        - 자식: 0
        - 실패: -1
        - code
            
            ```c
            #include <stdio.h>
            #include <unistd.h>
            
            int main(void) {
                printf("before fork\n");
                pid_t pid = fork();
            
                if (pid < 0) { // 실패 -1
                    perror("fork failed");
                    return 1;
                } else if (pid == 0) { // 자식
                    printf("child: pid=%d\n", getpid ());
                } else { // 자식의 PID를 받은 부모
                    printf("parent: pid=%d, child =%d\n", getpid (), pid);
                }
                printf("both: pid=%d\n", getpid ());
                return 0;
            }
            ```
            
    - fork, exec 분리 이유
        
        : 자식이 exec하기 전에 그 사이에 fd 조작해서 실행될 환경을 설정하려고
        
    - 부모 자식 fork 이후 상태와 흐름
    - printf, scanf 내부
        
        ```c
        // printf ("Hello, World!\n"); 와 동일한 역할:
        write(1, "Hello, World!\n", 14); // fd 1은 stdout(모니터)
        ```
        
3. exec() 동작
    
    : 현재 프로세스 메모리 새 프로그램으로 교체
    
    - execlp
        
        ```c
        #include <stdio.h>
        #include <unistd.h>
        
        int main(void) {
            printf("parent: start\n");
            pid_t pid = fork();
            
            if (pid == 0) { // 자식 프로세스: ls -l 실행
                printf("child: before exec\n");
                execlp("ls", "ls", "-l", NULL); // 첫 번째는 파일 이름, arv[0](실행할 프로그램 이름), argv[1](옵션), 인자 끝
                
                // exec 성공 시 주소 공간이 교체되므로 아래 코드는 절대 도달 불가!
                perror("exec failed");
            } else { // 부모 프로세스
                printf("parent: child=%d\n", pid);
                wait();
            }
            return 0;
        }
        
        ```
        
    - execvp
        
        ```c
        // #include 생략 (stdio, stdlib, unistd, string, sys/wait)
        int main(void) {
            printf("hello (pid:%d)\n", (int) getpid());
            int rc = fork();
            
            if (rc < 0) {
                fprintf(stderr, "fork failed\n");
                exit(1);
            } else if (rc == 0) { // 자식 프로세스
                printf("child (pid:%d)\n", (int) getpid());
                char *myargs[3];
                myargs[0] = strdup("wc"); // 프로그램 이름
                myargs[1] = strdup("p3.c"); // 대상 파일
                myargs[2] = NULL; // 끝을 알리는 NULL
                execvp(myargs, myargs); // wc p3.c 실행(실행 파일명, 인자 배열 전체)
            } else { // 부모 프로세스
                int wc = wait(NULL); // 자식이 끝날 때까지 대기
                printf("parent of %d (wc:%d)\n", rc, (int) getpid());
            }
            return 0;
        }
        ```
        
    
    (1) 메모리 교체 → exec() 호출한 원래 코드로 돌아올 수 없음
    
    cf. PID, open file entry 유지
    
    (2) PC를 새로운 main()으로
    
    - return 하지 X
    - 종류(l: list, v: vector, p=PATH 환경 변수 검색, e=직접 지정)
    - execvp: 인자를 배열(**`argv`**)로 넘기며, **`PATH`**를 검색하여 실행 파일을 찾음 ex. **execvp("ls", argv)**
4. fork → exec → wait 패턴
    - wait(&status):
        
        (1) 자식이 종료될 때까지 waiting 상태로 기다림 → 순서 보장
        
        (2) 자식의 종료 코드 회수(status에 자식의 종료 상태, 리턴은 자식의 PID)
        
        ```c
        // #include 생략
        int main(void) {
            pid_t pid = fork();
            
            if (pid == 0) {
                printf("child: working...\n");
                sleep(2);
                printf("child: done!\n");
                exit(42); // 자식은 종료 코드 42를 반환, exit() 호출해서 한 건 다 정상 종료(정상 종료 = ~강제종료)
            } else {
                int status;
                pid_t w = wait(&status); // 부모는 자식을 수거
                printf("parent: child %d exited\n", w);
                
                if (WIFEXITED(status)) {
                    printf("exit code: %d\n", WEXITSTATUS(status)); // 매크로로 42 추출
                }
            }
            return 0;
        }
        ```
        
        (3) 좀비 방지
        
        - wait 관련 함수
            - WIFEXITED(status): 정상 종료 여부 → true, false
                
                → 이게 참일 때만 WEXITSTATUS(status): 정상 종료일 때의 종료 코드
                
            - WIFSIGNALED(status): 시그널로 종료 여부
                
                → 이게 참일 때만 WTERMSIG(status): 시그널로 종료일 때의 시그널 번호(128+N)
                
        - exit(stauts)
            - 프로세스 종료하고, status에 상태 저장
            - 파일 닫고, 메모리 해제
            - SIGCHLD 보내고, 상태를 terminated로(PID, 종료 코드가 커널 프로세스 테이블에)
5. 좀비, 고아 → 수거(reap)
    - 좀비: 자식의 exit() 호출 → 종료 but 부모가 아직 wait() 안해서 종료 코드, PID 남아 있음 “not released”(= 부모가 살아 있는데 계속 wait() 안 하면 영구 좀비가 됨)
    - 고아: 자식이 부모보다 먼저 종료
        
        → init, systened 프로세스(PID = 1)가 입양 → 대신 wait()해서 좀비 방지
        
        collected and released
        
        cascading termination X
        
6. 셸, 파이프
    - 셸: 사용자의 명령을 읽고, 해당 프로그램을 실행해주는 프로세스
        
        ```c
        while (1) {
            printf("$ "); // 프롬프트 출력
            cmd = read_line(); // 사용자 입력 읽기
            args = parse(cmd); // 명령어를 토큰으로 분리
            
            if (is_builtin(args)) { // cd, exit 같은 내장 명령어 처리
                run_builtin(args); 
                continue;
            }
            
            pid_t pid = fork(); // 1. 자식 프로세스 생성
            
            if (pid == 0) {
                setup_redirections(cmd); // 2. 필요시 리다이렉션(fd 조작) 세팅
                execvp(args, args);   // 3. 새로운 프로그램으로 교체
                perror("exec failed");
                exit(1);
            } else {
                int status;
                waitpid(pid, &status, 0); // 4. 부모는 자식 작업이 끝날 때까지 대기
            }
        }
        ```
        
    - IO 리다이렉션(>): exec로 덮어쓰기 전 입출력 대상 바꾸기
        
        fork → close(1) -(쓰기 끝 닫음)→ open(file)-(제일 낮은 번호에 이 file 할당)→ exec
        
        ex. dup2(fd, STDOUT_FILENO) stdout을 fd로 교체
        
        - code
            
            ```c
            #include <stdio.h>
            #include <unistd.h>
            #include <string.h>
            #include <sys/wait.h>
            
            int main(void) {
                int fd[3];
                pipe(fd); // 파이프 생성: fd=read, fd[2]=write
                
                pid_t pid = fork();
                
                if (pid == 0) { // 자식 프로세스: 읽기 쪽만 사용
                    close(fd[2]); // 쓰기 파이프 닫기
                    char buf[4];
                    read(fd, buf, sizeof(buf)); // 파이프에서 데이터 읽기
                    printf("child got: %s\n", buf);
                    close(fd);
                } else { // 부모 프로세스: 쓰기 쪽만 사용
                    close(fd); // 읽기 파이프 닫기
                    char *msg = "hello from parent";
                    write(fd[2], msg, strlen(msg)+1); // 파이프에 데이터 쓰기
                    close(fd[2]);
                    wait(NULL); // 자식 대기
                }
                return 0;
            }
            ```
            
        
        ```c
        int rc = fork();
        
        if (rc == 0) {
            // 1. 표준 출력(fd 1)을 닫음
            close(STDOUT_FILENO); 
            
            // 2. 파일을 열면, 방금 비워진 가장 낮은 번호인 fd 1에 파일이 할당됨
            open("./p4.output", O_CREAT|O_WRONLY|O_TRUNC, S_IRWXU);
            
            char *myargs[1];
            myargs = strdup("wc");
            myargs[2] = strdup("p4.c");
            myargs[3] = NULL;
            
            // 3. wc 프로그램 실행. wc는 fd 1에 출력하려 하지만, 이제 fd 1은 파일이므로 파일에 써짐!
            execvp(myargs, myargs); 
        } else {
            wait(NULL);
        }
        ```
        
7. 프로세스 간 통신
    
    (1) 프로세스 종류
    
    - 협력 프로세스: 다른 프로세스랑 데이터를 주고받거나 영향을 주는 프로세스
    - 독립 프로세스: 반대
    
    (2) IPC 모델(프로세스 간 통신)
    
    |  | **Shared Memory** | **Message Passing** |
    | --- | --- | --- |
    | 통신 방식 | 공유 영역에 직접 읽고 씀 | send/receive 시스템콜 |
    | 커널 개입 | 설정 시 | 매 메시지 |
    | 속도 | 커널 우회 → 빠름 | 복사 오버헤드 → 느림 |
    | 동기화 | 프로그래머가 직접 | 커널 내장 |
    | 적합 | 대량 data, 같은 머신 | 소량 data, 분산 시스템 |
    | 예 | shm, mmap | pipe, socket |

    - Shared Memory
    
    ```c
    // --- main() 내부 ---
    const int SIZE = 4096;
    const char *name = "/OS";
    const char *msg0 = "Hello";
    const char *msg1 = "World!";
    
    // 공유 메모리 객체 생성
    int fd = shm_open(name, O_CREAT | O_RDWR, 0666);
    ftruncate(fd, SIZE); // 크기 설정
    
    // 주소 공간에 매핑
    char *ptr = (char *) mmap(0, SIZE, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
    
    // 메모리에 직접 쓰기
    sprintf(ptr, "%s", msg0);
    ptr += strlen(msg0);
    sprintf(ptr, "%s", msg1);
    ptr += strlen(msg1);
    소비자 (Consumer) 코드:
    // --- main() 내부 ---
    const int SIZE = 4096;
    const char *name = "/OS";
    
    // 기존 공유 메모리 객체 열기
    int fd = shm_open(name, O_RDONLY, 0666);
    
    // 주소 공간에 읽기 전용 매핑
    char *ptr = (char *) mmap(0, SIZE, PROT_READ, MAP_SHARED, fd, 0);
    
    // 메모리에서 직접 읽기
    printf("%s", (char *)ptr);
    
    // 사용 완료 후 삭제
    shm_unlink(name);
    ```
    
    - int fd = shm_open(name, O_CREAT | O_RDWR, 0666);
     : 공유 메모리 객체 생성, 열기 (공유 메모리라는 걸 열든지 만들고, 내 프로세스 FD 테이블 번호표 하나가 걔를 가리키게 해라)
    - mmap(): shm_open으로 만든 걸 프로세스 가상 주소 공간에 연결
    
    (3) pipe
    
    : 두 프로세스 간 단방향 통로
    
    ```c
    #include <stdio.h>
    #include <unistd.h>
    #include <string.h>
    #include <sys/wait.h>
    
    int main(void) {
        int fd[2];          // (0: 읽기, 1: 쓰기)
        pipe(fd);           // 커널이 fd[0]과 fd[1]에 진짜 FD 번호를 넣어줍니다(가장 작은 것 두 개)
    
        pid_t pid = fork();
    
        if (pid == 0) {     // --- 자식 프로세스 ---
            close(fd[1]);   // 2. 자식은 '쓰기'를 안 하니까 1번 상자의 번호를 닫습니다
            
            char buf[30];   
            int n = read(fd[0], buf, sizeof(buf) - 1); // 0번 상자에 든 번호로 읽기
            
            buf[n] = '\0';  // 문자열 끝을 표시
            printf("child got: %s\n", buf);
            
            close(fd[0]);   // 다 썼으니 '읽기' 번호도 닫습니다
        } 
        else {              // --- 부모 프로세스 ---
            close(fd[0]);   // 2. 부모는 '읽기'를 안 하니까 0번 상자의 번호를 닫습니다
            
            char *msg = "hello from parent";
            // 4. 1번 상자에 든 '쓰기' 번호로 데이터를 보냅니다
            write(fd[1], msg, strlen(msg)); 
            
            close(fd[1]);   // 다 보냈으니 '쓰기' 번호를 닫습니다
            wait(NULL);     // 자식이 죽을 때까지 기다려줍니다 (좀비 방지)
        }
        return 0;
    }
    ```

8. 시그널
    
    : 사건 발생을 알리는 비동기 메커니즘
    
    ```c
    #include <signal.h>
    #include <stdio.h>
    #include <unistd.h>
    
    // 시그널을 받으면 실행될 사용자 정의 핸들러 함수
    void handler(int sig) {
        printf("Caught signal %d\n", sig);
    }
    
    int main(void) {
        signal(SIGINT, handler); // SIGINT(Ctrl+C) 발생 시 handler 실행하도록 등록
        printf("PID=%d, waiting...\n", getpid());
        
        while (1) { 
            sleep(1); 
        }
        return 0;
    }
    ```
    
    - 종류
        
        
        | SIGCHLD | 자식 종료/정지 |
        | --- | --- |
        | SIGINT | 인터럽트 |
        | SIGTERM | 정상 종료 요청 |
        | SIGKILL | 강제 종료 요청(무시불가) |
        | SIGSTOP | 일시 정지(무시불가) |
        | SIGSEGV | 잘못된 메모리 접근 |
    - 전송 및 처리
        - 전송
            - kill(pid, SIG)
