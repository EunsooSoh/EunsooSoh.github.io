### 기본

- 문자열 탐색: 주어진 text string 에서 pattern string이 나타나는 모든 위치를 결정
- 보통
    - n: Text string 길이
    - m: Pattern string 길이
- 용어
    - 접두부: empty string과 첫 글자에서 시작하는 모든 문자열(진접두부: 자기 자신 제외)
    - 접미부:  empty string과 마지막 글자로 끝나는 모든 문자열(진접미부: 자기 자신 제외)
    

### Brute-force

- 코드
    
    ```cpp
    void BruteForce(string T, string P, int n, int m) {
    	for (int i = 0; i < n-m; i++) { // 패턴과 비교할 첫 인덱스
    		for (int j = 0; j < m; j++) {
    			if (P[j] != T[i+j]) break;
    		}
    		if (j == m) cout << "패턴이 텍스트의 "<< i << "번 째부터 나타남";
    	}
    }
    ```
    
    O(mn)
    

### Rabin-Karp

:  String 을 숫자 값으로 바꾼 뒤 해시 값을 계산해 매칭(패턴 전체를 하나의 십진수로 바꾸어 처리)

- 길이가 m인 pattern string P를 하나의 워드로 간주해 해시값 계산. 반복문 돌면서 i부터 길이가 m인 string에 대한 해시값 계산 및 비교
- Horner 방법: 이전 값에 d 곱하고 다음 항 더하기
- 코드

```cpp
void rabinKarp(string T, string P, int d, int q) { // q 는 해시함수를 위한 값, d는 진법(문자 집합 size로 하든지, 256같은 특정 수로 하든지)
	// d^(m-1) mod q 구하기
	int D = 1;
	for (int i = 0; i < m-1; i++) {
		D = D*d%q;
	}
	// 해시값 계산
	int h = 0; // h는 pattern string의 해시값
	int t = 0; // t는 text string의 해시값
	for (int i = 0; i < m; i++) {
		h = (d*h + P[i]) % q;
		t = (d*t + T[i]) % q; // 이건 인덱스 0부터 길이 m의 해시값 계산
	}
	// 해시값 비교
	for (int i = 0; i < (n-m+1); i++) {
		if (h == t) { // 해시값이 같을 때만
			int j;
			for (j = 0; j < m; j++) { // 전체 문자열 비교
				if (P[j] != T[i+j]) break;
			if (j == m) cout << "패턴이 위치 " << i << "에서 발생\n";
		}
		if (i < (n-m)) **t = (d*(t-T[i]*D) + T[i+m])%q;** // 다음 t값 계산
		// T[i]는 이제 빠질 문자, T[i+m]은 이제 추가할 문자
	}
}
```

최악: O(mn)

보통 O(m+n) - 패턴의 해시값과 텍스트의 해시값이 모두 불일치하는 경우

### KMP

: 

- 구해야 할 것: 접두사와 접미사가 일치하는 최대 길이
    
    ⇒ 접두사와 접미사가 일치하는 경우에 한해서 접프 가능 
    
- 접두사 따라가는 인덱스 j, 접미사 따라가는 인덱스 i가 있는데
    - 일치하면 i, j 같이 한 칸 옆으로
    - 불일치하면 i만 한 칸 옆으로 가고, j는 한 칸 뒤로 가서 걔가 가지는 인덱스 (SP배열 값)로 이동
- 코드

```cpp
void computeSP(string P, int SP[], int m) {
	SP[0] = -1; // 첫 번째 위치는 비교할 접두사나 접미사가 없으므로 -1로 초기화
	int j = -1; // 접두사/접미사가 일치하는 최대 구간의 끝 인덱스 기록할 변수
	for (int i = 1; i < m; i++) { // i: 현재 위치
		while(j >= 0 && P[j+1] != P[i]) j = SP[j]; // 현재 위치에서 접두사/접미사가 이어지지 않음 -> 더 짧은 접두사/접미사 후보로 점프
		if (P[j+1] == P[i]) j++; // 만약 접두사 다음 문자가 현재 문자와 같으면 일치 구간 하나 늘림
		SP[i] = j;
	}
}
// SP 배열: 패턴 문자열 인덱스 0~i까지의 접두사와 접미사가 일치하는 최대 길이(글자 수 -1)의미 

void KMP(string P, string T, int n, int m) { // n은 T의 길이, m은 P의 길이
	computeSP(P, SP, m);
	int j = -1; // j는 pattern 순회 
	for (int i = 0; i < n; i++) { // text string 순회
		while(j >= 0 && P[j+1] != T[i]) j = SP[j];
		if (P[j+1] == T[i]) j++;
		if (j == m-1) {
			cout << "패턴이 " << i-m+1 << "에서 일치함.\n"; // 그 문자열 다 돌아야 일치하는지 확인하니까 i-m+1임
			j = SP[j];
		}
	}
}
// 안 맞으면 pattern string 인덱스를 앞으로 땡겨서 확인 
```

: 패턴의 오른쪽부터 비교. 문자열 패턴과 문자열을 비교 시 문자열의 가장 뒷부분 위치를 비교하고 다르면 일정 길이만큼 이동해서 비교를 계속함

- 매칭은 왼쪽 → 오른쪽, 비교는 오른쪽 → 왼쪽
- 코드
    
    ```cpp
    void lastPos(string P, int m, int alpha[], int LP[]) {
    	for (int s : alpha) LP[s] = -1; // 초기화
    	for (int i = 0; i < m; i++) LP[P[i]-'a'] = i; // 각 알파벳이 가장 오른쪽에 있는 게 어디인지
    }
    
    void boyerMoore1(string T, int n, string P, int m, int alpha[]) {
    	lastPos(P, m, alpha, LP);
    	int i = 0;
    	int j; // 오른쪽에서 왼쪽으로 검사할 때 만나는 가장 오른쪽 불일치 위치
    	while(i <= n-m) {
    		for (j = m-1; j >= 0 && P[j] == T[i+j]; j--);
    		if (j == -1) { // 뒤에서부터 앞으로 끝까지 가고, 결국 -1에 도달 (다 일치했음)
    			cout << "패턴이 위치 " << i << "에서 발생\n";
    			i += m; // pattern string 길이만큼 i 이동
    		}
    		else i += (j - LP[T[j+i]-'a']); 
    	}
    }
    
    void goodSuffix(string P, int m, int skip[]) {
        // 1. 패턴 P를 뒤집어서 pr에 저장한다.
        string pr = P;
        reverse(pr.begin(), pr.end());
    
        // 2. 뒤집은 문자열 pr에 대해 접미사 매칭 길이(SP)를 계산한다 (접미사 배열 역할).
        computeSP(pr, SP, m);
    
        // 3. skip 배열을 최대 이동거리(m - 1 - SP[m-1])로 초기화해, 패턴이 최대한 멀리 이동할 기본값을 설정.
        for (int i = 0; i < m; i++) {
            skip[i] = m - 1 - SP[m - 1];
        }
    
        // 4. 각 위치 k에 대해, 패턴 접미사 부분과 일치하는 접두사를 찾아 더 작은 이동 거리로 skip 값을 갱신한다.
        for (int k = 0; k < m; k++) { // k는 접미사 길이별로 하나씩 살피는 변수
            int i = m - 1 - SP[k] - 1; //  접미사를 패턴 맨 앞에 맞춰 옮겼을 때 오른쪽으로 옮기는 칸 수, 접미사가 접두사로 가장 먼저 겹칠 수 있는 위치의 인덱스를 i에 저장
            if (skip[i] > k - SP[k]) 
                skip[i] = k - SP[k];
        }
    }
    
    void boyerMoore2(string T, int n, string P, int m, int alpha[]) {
    	lastPos(P, m, alpha, LP);
    	goodSuffix(P, m, skip);
    	int i = 0;
    	while(i <= n-m) {
    		int j;
    		for (j = m-1; j	>= 0 && P[j] == T[i+j]; j--); // 끝나면 text랑 pattern이 불일치하는 j가 됨
    		if (j == -1) {
    			cout << "패턴이 위치 " << i << "에서 발생\n";
    			i += skip[j-1];
    		}
    		else i += max(skip[j], j - LP[T[j+i]]); // 그니까 lastPos로 넘길 수 있었던 거랑 skip 배열로 넘길 수 있는 것 중 큰 걸로 넘김
    	}
    }	
    ```
