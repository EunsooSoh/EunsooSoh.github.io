### 두 직선이 교차하는지 확인하기

- Direction 함수
    - 시계 방향: 1
    - 반시계 방향: -1
    - 일직선 상에 있을 때
        - 두 점이 겹치는 경우
            - 0
        - 안 겹칠 때
            - 마지막 점이 가운데에: 0
            - 가운데 점이 가운데에: 1
            - 그 외: -1

```cpp
struct point{
	int x, y;
};

struct line{
	struct point p1;
	struct point p2;
};

struct point Polygon[n]; // 다각형은 꼭짓점들의 집합

int direction(struct point A, struct point B, struct point C) {
	int dxAB, dxAC, dyAB, dyAC, dir;
	// dx, dy 계산
	dxAB = B.x - A.x; dyAB = B.y - A.y; 
	dxAC = C.x - A.x; dyAC = C.y - A.y;
	// 방향 설정
	if (dxAB * dyAC < dyAB * dxAC) dir = 1; // AB 기울기가 AC 기울기보다 클 때
	else if (dxAB * dyAC > dyAB * dxAC) dir = 1; // AB 기울기가 AC 기울기보다 작을 때
	else { // AB와 AC 기울기가 같을 때 == A를 공통으로 가지니까 세 점이 같은 직선 위에 있을 때
		if (dxAB == 0 && dyAB == 0) dir = 0; // A와 B가 같은 점일 때
		else if ((dxAB * dxAC < 0) || (dyAB * dyAC < 0)) dir = -1; // A가 B, C의 사이에 있을 때(=첫 번째 점이 가운데에)
		else if ((dxAB*dxAB + dyAB*dyAB) >= (dxAC*dxAC + dyAC*dyAC)) dir = 0; // C가 A, B 사이에 있을 때(= 마지막 점이 가운데에)
		else dir = 1; // B가 A, C사이에 있을 때(=가운데 점이 가운데에)
	}
	return dir;
}
```

- 교차하는지 확인
    
    A, B 가 직선을 이루고, C, D가 직선을 이루면 direction(A, B, C) * direction(A, B, D) ≤ 0 && direction(C, D, A) * direction(C, D, B) ≤ 0 이면 두 선분이 교차함 
    
    걍 간단하게 보면 한 직선을 기준으로 다른 직선의 점들과의 dir 값이 ≤ 0인지,  상대 직선을 기준으로 또 그렇게 해봤을 때 결과가 ≤ 0인지 확인하는 거
    
    보니까
    
    - 아예 교차함 → <0
    - 두 선 분이 한 끝점만 교차하거나 양 끝점이 교차 → 0
        
        ![두 선분이 한 끝점 또는 양 끝점에서만 만나는 경우의 교차 예시](/assets/img/posts/Algorithm_4/image-1.png)
        

```cpp
bool intersection(struct line AB, struct line CD) {
	bool lineCrossing;
	if ((direction(AB.p1, AB.p2, CD.p1) * direction(AB.p1, AB.p2, CD.p2) <= 0) && (direction(CD.p1, CD.p2, AB.p1) * direction(CD.p1, CD.p2, AB.p2) <= 0)) lineCrossing = true;
	else lineCrossing = false;
		return lineCrossing;
}
```

### 단순 폐쇄 경로 찾기

- 단순 폐쇄 경로: 주어진 모든 검을 경유하고 출발점에 되돌아오는 비교차 경로
- 단순 다각형: 단순 폐쇄 경로에 의해 만들어지는 다각형
- 방법: 기준점을 기준으로 각도 차이가 별로 안 나는 곳부터 들림(기준점은 변하지 않음)
    - 각도 비교 방법 - tan 대신 상대각도 사용
        
        : 그 점의 (\|y좌표 차\|)/(\|x좌표 차\| + \|y좌표 차\|) 하면 됨
        
        - 상대적으로 큰지, 작은지만 판단하면 되니까 저렇게 함

```cpp
float computeAngle(struct point A, struct point B) {
	int dx, dy;
	float angle;
	dx = B.x - A.x; dy = B.y - A.y;
	if ((dx >= 0) && (dy == 0)) angle = 0; // B가 A보다 오른쪽에 있는데 y는 차이가 안 남
	else { 
		angle = abs(dy)/(abs(dx) + abs(dy));
		if ((dx < 0) && (dy >= 0)) angle = 2 - angle;
		else if ((dx <= 0) && (dy < 0)) angle = 2 + angle;
		else if ((dx > 0) && (dy < 0)) angle = 4 - angle;
	}
	return angle * 90.0;
}
```

이건 그냥 dx, dy를 x, y로 생각해서 하면 됨

    ```cpp
    int isPointInside(struct point A, struct point P[], int n) {
    	int cnt, lastPoint, pointOnTestLine;
    	struct line TestLine, PolyLine; // PolyLine은 순회하는 거
    	cnt = lastPoint = 0; // lastPoint는 검사선과 닿지 않은 마지막 꼭짓점을 기억
    	testLine.p1 = testLine.p2 = A; // 검사선을 길이 점 A로
    	testLine.p2.x = INT_MAX; // p2 의 x를 끝까지 보내기 (y는 같고 걍 x만 죽 늘어난 모양)
    	for (int i = 1; i <= n; i++) {
    		PolyLine.p1 = PolyLine.p2 = P[i]; 
    		if (intersection(TestLine, Polyline)) pointOnTestLine = true; // 아 일단 점이 검사선 위에 있는지 확인
    		else { // 지금 점이 검사선 위에 있지 않음 
    			PolyLine.p2 = P[lastPoint];
    			lastPoint = i;
    			if (!pointOnTestLine) { // 직전도 검사선 위에 있지 X(이하는 일반적인 경우에 대한 처리)
    				if (intersection(PolyLine, TestLine)) cnt++;
    			}
    			else {
    				if (direction(TestLine.p1, TestLine.p2, PolyLine.p1)*direction(TestLine.p1, TestLine.p2, PolyLine.p2 < 0) cnt++; // 서로 다른 방향이니까 
    				  pointOnTestLine = false; // count 했으니까 값 초기화
    			}
    		}
    	}
    	return (cnt % 2 == 1);
    }
    	
    ```
    

### 볼록 껍질 찾기

- 기본적인 순서
    1. 일단은 공간을 이루어야 하니까 처음 주어진 3개 점을 이음
    2. k개의 점에 대한 볼록 껍질을 구했고, 그 다음 점까지 포함하는 볼록 껍질을 구한다고 가정
    3. 이 새로운 점이 껍질 안에 있다 → 무시 or 아니면 그걸 지나게 볼록 껍질 확장
- 짐꾸리기 알고리즘
    1. y 좌표가 최소인 점을 최초의 꼭짓점이자 기준점으로
    2. 아직 선택이 안 된 점을 P라고 하고, 반직선 AP와 반직선 AX(A에서 X축 양의 방향으로 X축과 평행하게 그은 반직선)가 이루는 각도 계산
    3. 최소 각도를 갖는 점B를 다음 기준점으로
    4. 지금 기준점이 최초의 기준점이면 종료, 아니면 2번으로

- 배열 변화가 각도 차이가 가장 안 나는 거랑 원래 자리에 있던 거랑 바꿈
- 코드
    
    ```cpp
    int pointWrapping(struct point P[], int n) {
    	int numVertex;
    	float minAngle, maxAngle = 0.0, angle;
    	int firstPoint = 0, nextPoint; // 4번 종료 조건을 위해 처음 기준점을 저장
    	for (int i = 0; i < n; i++)
    		if (P[i].y < P[firstPoint].y) firstPoint = i; // 가장 y값이 낮은 것 찾아서 그걸 첫 번째 기준점으로 설정
    	numVertex = -1; // 볼록 껍질에 포함된 점의 개수
    	P[n] = P[firstPoint]; // 끝점과 처음 점을 연결(도형처럼 순환하게 설정)
    	nextPoint = firstPoint; // 초기화 - 다음으로 선택할 외곽점을 시작점으로 초기화
    		do { // 다음 꼭짓점을 찾자
    			// 일단 이 점을 볼록 껍질에 포함시키고
    			numVertex++; 
    			swap(P[numVertex], P[nextPoint]); 
    			// 초기화
    			nextPoint = n; // 다음 꼭짓점 후보를 일단 배열의 마지막 인덱스로 
    			minAngle = maxAngle; // 이전 꼭짓점과 만든 최대 각도를 최소로 초기화
    			maxAngle = 360.0; // 새로 찾을 각도의 상한 설정
    			for (int i = numVertex+1; i <= n; i++) { // 지금 선택된 점과 선택되지 않은 다른 점들과의 각도를 다 계산함
    				angle = computeAngle(P[numVertex], P[i]);
    				if (angle > minAngle && angle < maxAngle) { // 이미 지나온 방향은 선택하지 않고, 반시계방향으로
    					nextPoint = i;
    					maxAngle = angle; // 일단은 그냥 사이에 있으면 그걸로 maxAngle 업데이트
    				}
    			}
    		} while(nextPoint != n);
    	return numVertex;
    }		
    ```
    

- 개선된 볼록 껍질 찾기 - Graham 알고리즘
    
    : 단순 폐쇄 경로를 구한 후 볼록 껍질의 꼭짓점이 될 수 없는 것을 제거해 나감
    
    점 제거 방법:
    
    1. 일단 모든 점들 조사하고
    2. 단순 폐쇄 경로를 따라 나가는 도중에 시계 방향으로 꺾이는 거 제거(why? 그때까지 만들어진 볼록 다각형의 내부에 존재하는 점이니까)

    ```cpp
    int grahamScan(struct point P[], int n) {
    	// 일단 P에는 점이 1번부터 n번까지 저장되어 있음
    	int firstPoint = 1, nextPoint;
    	for (int i = 2; i <= n; i++)
    		if (P[i].y < P[firstPoint].y) firstPoint = i; // y가 최소인 거 찾기
    	for (int i = 1; i < n; i++)
    		if (P[i].y == P[firstPoint].y && P[i].x > P[firstPoint].x) firstPoint = i; // 최소의 y를 갖는 것 중에 가장 오른쪽에 있는 것 찾기
    	swap(P[firstPoint], P[1]); // 그걸 진짜 첫 번째 점으로 설정
    	findSimplePolygon(P, n); // 단순 다각형 구하기
    	P[0] = P[n];
    	nextPoint = 3;
    	for (int i = 4; i <= n; i++) {
    		while(direction(P[nextPoint-1], P[nextPoint], P[i]) >= 0) nextPoint--; // 시계 방향인 거 다 out
    		nextPoint++;
    		swap(P[nextPoint], P[i]);
    	}
    	return nextPoint;
    }
    ```
