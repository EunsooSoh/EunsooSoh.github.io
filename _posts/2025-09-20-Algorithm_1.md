## 기본 내용

- 정렬 시간 = 레코드 접근 시간 + 레코드 비교/재배치 시간
- 제자리성: 추가로 요구하는 공간 복잡도가 최대 O(1)인 알고리즘
- 안정성: 동일한 키에 대하여 정렬 이후 원본 레코드의 순서 유지

## 비교 기반 정렬 방법

### 선택 정렬

- 코드
    - 뒷 리스트에서 가장 작은 수를 앞 수와 바꿈. ⇒ 앞에 정렬된 것들이 쌓임.
    
    ```cpp
    void selectionSort(int list[], int n) {
    	int minIndex;
    	for (int i = 0; i < n-1; i++) {
    		minIndex = i;
    		for (int j = i + 1; j < n; j++) {
    			if (list[minIndex] > list[j]) minIndex = j;
    		}
    		**if (minIndex != i)** swap(list[minIndex], list[i]);
    	}
    }
    ```
    
    - py
        
        ```python
        def selectionSort(list, n):
        	for i in range(n):
        		minIndex = i
        		for j in range(i+1, n):
        			if list[minIndex] > list[j]:
        				minIndex = j	
        		list[minIndex], list[i] = list[i], list[minIndex]
        ```
        
- 불안정
- 제자리
- 평균, 최악 $O(n^2)$
    - 입력 자료 순서와 무관하게 항상 똑같은 횟수 비교

### 삽입 정렬 !! 질문드리기 왜 -무한대로 0을 초기화 해두는지? (for 문 안 쓰고 while 쓸 때를 위해)

- 코드
    - 바로 뒤에 있는 것을 앞쪽 정렬된 리스트의 적절한 곳에 삽입함
    
    ```c
    void insertionSort(int list[], int n) {
    	int i, j, value;
    	for (i = 2; i < n; i++) {
    		value = list[i];
    		j = i;
    		while (list[j-1] > value) { // 내 앞에 있는 게 나보다 큼(안 맞음)
    			list[j] = list[j-1]; // 맞을 때까지 뒤로 당김
    			j--;
    		}
    		list[j] = value;
    	}
    }
    ```
    
    - py

- 평균, 최악(역순): $O(n^2)$

### 쉘 정렬

- 코드 
    - 삽입 정렬을 띄엄띄엄
    
    ```cpp
    void ShellSort(int list[], int n) {
    	int gap = 1; 
    	int value;
    	
    	while(gap < n/3) gap = 3*gap+1; // 해당 수열 중 gap이 가장 클 수 있게
    	
    	for(; gap >= 1; gap /= 3) { // gap을 3씩 나눠가는 중(gap 크기 줄이기)
    		for(int i = gap; i < n; i++) {
    			value = list[i]; // 현재 값 추출
    			int j = i; // 현재 인덱스 추출
    			while(j >= gap && list[j-gap] > value) { // 현재 값(value) 한 gap 앞에 있는 것보다 작음
    				list[j] = list[j-gap];
    				j -= gap;
    			}
    			list[j] = value; // 앞에 있는 원소에 value를 넣음
    		}
    	}
    }
    ```
    
    

### 퀵 정렬

- 코드
    
    대상이 주어지면 그 대상에서 피벗 설정
    
    → 그거 기준으로 나머지 숫자 위치 이동
    
    → 그 피벗의 위치는 결정되어 고정됨!
    
    → 그래서 이 동작을 모든 숫자에 대해서 하면 전체가 정렬됨
    
    - py
        
        ```python
        def partition(a, left, right):
        	low = left
        	high = right + 1
        	pivot = a[left]
        	
        	while True:
        		while True:
        			low += 1
        			if (low > right) or (a[low] >= pivot):
        				break
        		while True:
        			high -= 1
        			if (high < low) || (a[high] <= pivot):
        				break
        		if low >= high:
        			break
        		a[low], a[high] = a[high], a[low]
        	a[left], a[high] = a[high], a[left]
        	return high
        	
        	
        def quikSort(a, left, right):
        	pivot = partition(a, left, right)
        	quikSort(a, left, pivot-1)
        	quikSort(a, pivot + 1, right)
        ```
        
    
    ```cpp
    int partition(int list[], int left, int right) {
    	int partElem = left; // pivot 인덱스
    	int value = list[partElem];
    	
    	do{
    		do while(list[++left] < value);
    		do while(list[--right] > value);
    		// 교차가 되지 않았다 == 올바른 위치에 있지 않다
    		if (left < right) swap(list[left], list[right]);
    		else break; // 교차가 됐다 == (피벗 제외)다 올바른 위치에 있다 == (피벗 제외)정렬되어 있다
    	}while(1);
    	list[partElem] = list[right]; // 현재 피벗 자리에 있는 값을 첫 번째로 보냄
    	list[right] = value; // 현재 피벗 자리에 올바른 값 저장
    	return right; // pivot 위치 반환
    }
    	
    	
    void quikSort(int list[], int left, int right) {
    	if(right > left) {
    		int pivot = partition(list, left, right);
    		quikSort(list, left, pivot-1);
    		quikSort(list, pivot+1, right);
    	}
    }
    ```
    
- 불안정
- 제자리(단, 재귀형태라 복귀주소가 계속 시스템 스택에 쌓이긴 함)
- 최선, 평균 $O(n*log_2n)$:  거의 균등한 리스트로 분할
- 최악$O(n^2)$:  극도로 불균등한 리스트로 분할(예: 이미 정렬된 리스트, 피벗이 최소 또는 최대가 되어서)

### 합병 정렬

- 코드
    
    일단 중간 기준으로 나누고 합칠 때 맞게 따닥 넣기
    
    ```cpp
    #define NUM_OF_KEYS 10 // 정렬하려는 리스트의 레코드 개수
    
    void merge(int list[], int left, int mid, int right) { // right도 인덱스임
    	int buffer[NUM_OF_KEYS];
    	// 왼쪽 리스트: left~mid, 오른쪽 리스트: mid+1~right
    	int leftPtr = left, rightPtr = mid+1, bufPtr = left;
    	while(leftPtr <= mid && rightPtr <= right) { // 둘 중 하나의 리스트가 끝나지 않는 동안
    		if (list[leftPtr] < list[rightPtr]) buffer[bufPtr++] = list[leftPtr++];
    		else buffer[bufPtr++] = list[rightPtr++];
    	}
    	if (leftPtr <= mid) { // 왼쪽 리스트에 있는 게 아직 남음
    	// 등호 붙음! leftPtr 자체가 왼쪽 리스트에서 옮길 다음 인덱스를 의미
    		for (int i = leftPtr; i <= mid; i++) buffer[bufPtr++] = list[i];
    	}
    	else {
    		for (int i = rightPtr; i <= right; i++) buffer[bufPtr++] = list[i];
    	}
    	// 마지막으로 원래 리스트에 전체 복사
    	for (int i = left; i <= right; i++) list[i] = buffer[i];
    }
    void mergeSort(int list[], int left, int right) {
    	if (left < right) {
    		int mid = (left + right)/2;
    		mergeSort(list, left, mid);
    		mergeSort(list, mid + 1, right); 
    		merge(list, left, mid, right);
    	}
    }
    ```

### 힙 정렬

- 코드 
    
    heap에 정렬할 것들 하나씩 삽입 → heap에서 하나씩 빼기
    
    ```cpp
    #include <iostream>
    using namespace std;
    
    void makeHeap(int list[], int root, int n) { // n 은 노드 개수
    	int parent = root;
    	int val = list[parent]; // 현재 값
    	int left = 2*parent;
    	int right = left + 1;
    	int child;
    	
    	while(left <= n) {
    		// 좌, 우 중 큰 자식의 인덱스를 child에 저장
    		if (right <= n && list[left] < list[right]) { // 오른쪽 자식이 존재 & 좌, 우 중 더 큰 것 택1
    			child = right;
    		}
    		else child = left;
    		
    		if (val < list[child]) { // 힙 조건이 안 맞음
    			list[parent] = list[child]; // 나보다 큰 내 자식을 위로 올림
    			parent = child; // 나 인덱스 밑으로
    			// 자식들도 인덱스 바꿔
    			left = parent * 2;
    			right = left + 1;
    		}
    		else break; // 맞음
    	}
    	list[parent] = val;
    }
    
    void heapSort(int list[], int n) {
    	// 힙 만들기
    	for(int i = n/2; i > 0; i--) { // 마지막 부모 노드에서부터 1번까지(최상위 노드)
    		makeHeap(list, i, n); // i에서 끝까지가 힙이 되게
    	}
    	for(int i = n; i > 1; i--) { // 하나씩 꺼내서 가장 큰 값을 맨 뒤로 보냄
    		swap(list[1], list[i]); // 제일 큰 값을 현재 맨 뒤(i) 번째 자리와 바꿈
    		makeHeap(list, 1, i-1); // 넘긴 가장 큰 값만 빼고 앞 부분을 다시 힙 구조로
    	}
    }
    
    int main() {
        int list[] = {0, 8, 4, 5, 6, 7, 2 ,3, 1};
        heapSort(list, 8);
        
        for(int i = 1; i < 9; i++) cout << list[i] << ' ';
    }		
    ```

    		cnt[list[i]]--;
    	}

### 버킷 정렬

- 코드
    - 자리수 대로 버킷에 넣어서 정렬
    
    ```cpp
    #include <iostream>
    #include <vector>
    using namespace std;
    
    // 삽입 정렬 함수
    
    void bucketSort(float arr[], int n) {
        vector<float> buckets[n];
        // 1. 데이터 분배
        for (int i = 0; i < n; i++) {
            int idx = n * arr[i]; // n개 버킷에 분배
            buckets[idx].push_back(arr[i]);
        }
        // 2. 각 버킷 삽입 정렬
        for (int i = 0; i < n; i++) {
            insertionSort(buckets[i]);
        }
        // 3. 버킷 합치기
        int idx = 0;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < buckets[i].size(); j++) {
                arr[idx++] = buckets[i][j];
            }
        }
    }
    
    int main() {
        float arr[] = {0.78, 0.17, 0.39, 0.26, 0.72, 0.94, 0.21, 0.12, 0.23, 0.69};
        int n = sizeof(arr)/sizeof(arr[0]);
        bucketSort(arr, n);
        cout << "Sorted Output: ";
        for (int i = 0; i < n; i++) cout << arr[i] << " ";
        cout << endl;
        return 0;
    }
    ```
    
- 평균 $O(n)$: 레코드가 균등하게 분포(그니까 각 버킷마다 지나치게 긴 게 없음)
- 최악$O(n^2)$:  한 버킷만 길어지는 게 있음
