### 순차 탐색

- 코드1
    
    ```cpp
    int seqSearch(int list[], int n, int key) {
    	int i = 0;
    	while((i < n) && (list[i] != key)) i++;
    	return i;
    }
    // return 된 인덱스가 n이면 탐색 실패
    ```
    
- 코드2
    
    ```cpp
    typedef struct node{
    	int key;
    	struct node *link;
    } node;
    
    node *seqSearch(int key) {
    	node *p;
    	p = head->link; // 헤드가 더미라서 head->link가 첫 번째 노드임
    	while((p!=NULL) && (p->key != key)) t = t->link;
    	return t;
    }
    ```
    
- 평균, 최악 $O(n)$

### 이진 탐색

- 입력 레코드가 반드시 정렬되어 있어야 함.
- 코드(반복문으로 구현)
    
    ```cpp
    int biSearch(int list[], int n, int key) {
    	int low = 0;
    	int high = n-1;
    	
    	while(low <= high) {
    		int mid = (low + high)/2;
    		if (list[mid] == key) return mid;
    		else if (list[mid] < key) low = mid+1;
    		else high = mid-1;
    	}
    	return -1; // 탐색 실패
    }
    ```
    

### 이진 탐색 나무(BST)

- 기본적인 구조
    
    ```cpp
    typedef struct node {
    	struct node *left;
    	int key;
    	struct node *right;
    } node;
    
    void TreeInit() {
    	node *p, *head;
    	head = (node *)malloc(sizeof(node));
    	head->left = head->right = NULL;
    	head->key = 0;
    }
    ```
    
- 탐색 코드
    
    ```cpp
    node *TreeSearch(node *head, int key) {
    	node *t = head->right; // 트리의 head는 더미이고, 무조건 right 자식을 최상위 노드로
    	while(p != NULL) {
    		if (p->key == key) return p;
    		else if (p->key < key) p = p->right;
    		else p = p->left;
    	}
    	return NULL; // 탐색 실패
    }
    ```
    
- 삽입 함수 코드
    
    ```cpp
    void TreeInsert(node *head, int key) { // key: 삽입 키
    	node *p, *t;
    	p = head;
    	t = p->right;
    	
    	while(t != NULL) {
    		p = t;
    		if (t->key == key) return; // 이미 있으니까 그냥 안 넣음
    		else if (t->key > key) t = t->left;
    		else t = t->right;
    	}
    	
    	t = (node *)malloc(sizeof(node));
    	t->left = t->right = NULL;
    	t->key = key;
    	
    	if (key < p->key) p->l = t;
    	else p->r = t;
    }
    ```
    
- 삭제 함수 코드
        
    ```cpp
    TreeDel(node *head, int key) {
        node *p, *c, *t, *x;
        p = head; x = head->right;
        x = TreeSearch(x, key); // key에 해당하는 원소 추출(x가 내가 지울 거)
        if (x == NULL) return;
        
        else t = x; // t는 x와 같은 노드를 가리킴
        
        // 삭제할 노드의 부모에 연결해줄 x를 설정하는 단계
        if (t->right == NULL) x=t->left; // case1: 삭제할 게 우자식 없음
        else if (t->right->left == NULL) { // case2: 삭제할 거의 우자식의 좌자식이 없음
        // 일단 x는 계속 변하고, t는 원래 x노드를 가지고 있는 거라고 생각
            x = t->right;
            x->left = t->left;
        }
        else { // 그 외
            c = x->right;
            whlie(c->left->left != NULL) c = c->left; // c가 오른쪽 서브트리의 가장 왼쪽 자식의 부모
            x = c->left;
            c->left = x->right;
            x->left = t->left; x->right = t->right;
        }
        // 부모와 연결해줄 것 주변의 위치 조정 이후 t해제
        free(t);
        
        // 부모와 x를 연결하는 단계 
        if (key < p->key) p->l = x;
        else p->r = x;
    }	
    ```
        
    
    ```cpp
    void TreeDel(node *head, int key) {
        node *p, *x, *c, *t;
        p = head; x = head->right;
    
        // 1. 삭제할 노드(x)와 부모(p) 찾기
        while (x != NULL && x->key != key) {
            p = x;
            if (key < x->key) x = x->left;
            else x = x->right;
        }
        if (x == NULL) return; // 삭제할 노드 없음
    
        t = x; // 삭제 대상 노드
    
        // 2. 삭제할 노드의 대체 노드(x) 결정
        if (t->right == NULL) {
            x = t->left; // case 1: 오른쪽 자식 없음
        } else if (t->right->left == NULL) {
            x = t->right; // case 2: 오른쪽 자식의 왼쪽 자식 없음
            x->left = t->left;
        } else {
            c = t->right;
            while (c->left->left != NULL) c = c->left;
            x = c->left;
            c->left = x->right;
            x->left = t->left;
            x->right = t->right;
        }
    		
    		// 3. 삭제 대상 노드 메모리 해제
        free(t);
        
        // 4. 부모와 x를 연결
        if (key < p->key) p->left = x;
        else p->right = x;  
    }
    ```
    

- BST 단점: 경사나무 형성 → 탐색 시간 복잡도 O(n)

⇒ 균형 나무를 만들자! (2-3-4 tree, red-black tree)

- 균형 나무: 좌우가 같은 높이 유지
    - 트리에서 깊이와 높이
        - 깊이: 현재 노드까지의 조상 노드의 개수
        - 높이: 트리의 최대 레벨
            - 레벨은 0부터 시작

### 2-3-4 나무

- 삽입/탐색/삭제 시 요구되는 시간 복잡도 **O(logn)**
- k-노드: 링크 k개, 키 값 k-1개를 가진 노드

- 연산
    - 삽입: 그냥 맞는 자리에 넣으면 됨 ⇒ 추가 가지 생성 없이 **기존 노드와 합류**
        - 예외: 이미 4개 차 있음
    - 분할: 가운데에 있는 거를 위로 올리고 양 옆에 있는 걸 쪼갬
        - 탐색을 통해서만 수행 ⇒ 탐색되지 않은 부분의 4-노드는 다음 탐색까지 분할되지 않음
        - 최대 분할 노드 수 = 나무 높이 (다 4노드인 경우)

- 근데 복잡해

⇒ RBT를 만들어 보자!

### Red-Black Tree

: 2-3-4 나무를 이진 탐색 트리 형태로 표현한 자료구조 (이진 탐색 트리의 조건을 모두 만족해야 함)

- 이진 탐색 트리의 조건
    - 최대 두 개의 자식 (→ 구현 용이)
    - 서브 트리 간의 순서 존재
    - 노드 개수 n
        - 간선 개수 n-1 (루트 빼고 다 부모 한 명이라)
        - NULL 링크 수: 2n(전체 링크 수) - (n-1) (간선 개수) = n + 1

색(Tag: 자기 노드로 들어오는 링크의 색)

- 흑: 보통
- 적: 3-노드 or 4-노드 내 레코드 연결

! 동일한 2-3-4 나무에 대해 여러 개의 RBT 표현이 가능

- 성질
    - 어떤 노드의 태그가 적색 → 자식의 태그는 무조건 흑색 ⇒ 경로 상에 두 개의 적색 링크가 연달아 나타나지 않음 ⇒ 블랙 노드만이 레드 노드의 부모가 될 수 있음
    - 어떤 노드에서 그 자식의 모든 잎 노드까지의 단순 경로 상에 존재하는 흑색 링크의 개수는 동일
        - 단순 경로: 한 노드에서 다른 노드까지 한 번씩만 각 노드를 지나면서 도달하는 길
        - 그래서 최단 거리는 흑노드만 지나는 것임
- 유튜브 강의 내용 정리([https://www.youtube.com/watch?v=2MdsebfJOyM](https://www.youtube.com/watch?v=2MdsebfJOyM))
    
    ### 삽입
    
    1. 삽입 전 RB 트리 속성 만족
    2. 삽입 방식은 BST와 동일
    3. 삽입 후 RB 트리 위반 여부 확인
    4. RB 트리 속성을 위반했으면 재조정
    5. RB 트리 속성을 다시 만족  
    
    ⇒ 즉, 일단 red로 넣고 조정
    
- 구현
    
    ```cpp
    typedef struct node {
        int key, tag; // 1: 적색, 0: 흑색
        struct node *left, *right;
    } node;
    
    // 전역 포인터
    node *head, *tail, *x, *p, *g, *gg;
    
    void RB_init() {
        tail = (node *)malloc(sizeof(node));
        tail->left = tail->right = tail;
        tail->tag = 0;
    
        head = (node *)malloc(sizeof(node));
        head->key = 0;
        head->tag = 0;
        head->right = tail;
    }
    
    node* rotate(int k, node *y) { // k는 트리에 새롭게 들어가는 노드의 키값
        node *u, *v; // 회전 대상 노드
        u = (k < y->key) ? y->left : y->right;
        if (k < u->key) {
            v = u->left; u->left = v->right; v->right = u;             // 우회전
        } 
        else {
            v = u->right; u->right = v->left; v->left = u;             // 좌회전(우회전에서 오른쪽<=>왼쪽 하면 된다)
        }
        if (k < y->key) y->left = v; // y가 새로 들어가는 거의 부모인가봄
        else y->right = v;
        return v;
    }
    
    void split(int k) { // 빨간색 연달아 오는 것 방지
    		// 현재 노드를 빨간색으로 칠하고 자식들은 검은색으로
        x->tag = 1;
        x->left->tag = 0;
        x->right->tag = 0;
        // 부모도 빨간색이면
        if (p->tag == 1) {
            g->tag = 1; // 조부모를 빨간색으로 칠하고
            if ((k < g->key) != (k < p->key)) { // 지그재그형이면
                rotate(k, g); // 조부모기준 회전
                p = x; // 원래 부모가 있던 자리에 내가 간다? 
            }
            rotate(k, gg);
            x = p; //?
        }
        x->tag = 0;
        head->right->tag = 0;
    }
    
    void RB_insert(int k) {
        x = head->right; p = head; g = head;
        while (x != tail) {
            gg = g; g = p; p = x;
            if (k < x->key) x = x->left;
            else x = x->right;
            if ((x->left->tag == 1) && (x->right->tag == 1))
                split(k);
        }
        x = (node *)malloc(sizeof(node));
        x->key = k; x->left = x->right = tail; x->tag = 1;
        if (k < p->key) p->left = x;
        else p->right = x;
        split(k);
    }
    
    ```
    
    - tree 비어 있어도 head, tail 노드 존재함
- 코드
    - 기본 구조 및 초기화
    
    ```cpp
    typedef struct node{
    	int key, tag;
    	struct node *l, *r;
    }node;
    
    node *head, *tail, *x, *p, *g, *gg;
    
    void RB_init() {
    	tail = (node *)malloc(sizeof(node));
    	tail->l = tail->r = tail; // 자시 자신을
    	tail->tag = 0;
    	
    	head = (node *)malloc(sizeof(node));
    	head->key = head->tag = 0;
    	head->r = tail;
    }
    // 일단 tail은 자기 자신을 가리키는 노드로서 존재함.
    // 이게 연결 리스트 꼬리 같은 게 아니라 리프노드들이 다 이걸 가리킴
    // 지금은 아무것도 없으니까 head->r (첫 번째 노드)가 tail이 된다
    ```
    
    - 회전 연산
    
    ```cpp
    void rotate(int k, node *y) {
    	node *u, *v;
    	// 회전 기준 설정
    	u = (k < y->key)? y->l:y->r;
    	// 회전 방향 설정
    	if (k < u->key) { // 내 거가 더 작으니까 우회전(이건 걍.. 사례 하나 고정해서 외우자)
    		v = u->l;
    		u->l = v->r; // 타고 내려간다고 이해 
    		v->r = u;
    	}
    	else { // 반대는 방향만 바꾸면 됨
    		v = u->r; u->r = v->l; v->l = u;
    	}
    	// 부모 연결
    	if (y->key > key) y->l = v; else y->r = v;
    }
    ```
    
    - 분할 연산
    
    ```cpp
    void split(int k) {
    	x->tag = 1; x->l->tag = x->r->tag = 0; // 나 빨, 내 자식 검
    	if (p->tag == 1) { // 부모도 빨간색 !
    		g->tag = 1; // 조부모 빨간색으로 바꾸고
    		if ((k < g->key) != (k < p->key)) { // 쫙 펴져 있지 않음(지그재그임)
    			rotate(k, g); // 부모와 나를 회전
    			p = x;
    		}
    		rotate(k, gg); // 부모와 조부모 회전
    		x = p;
    		x->tag = 0; // 나를 검은색으로
    	}
    	head->r->tag = 0;
    }
    ```
    
    - 삽입 연산
    
    ```cpp
    void RB_insert(int k) {
    	x = head->r; p = head; g = head;
    	while(x != tail) {
    		gg = g; g = p; p = x; // 위지 갱신
    		if (k < x->key) x = x->l; else x = x->r; // x를 올바른 위치로
    		// x자식들이 다 빨간색이면 split
    		if ((x->l->tag == 1) && (x->r->tag == 1)) split(k); 
    	}
    	// 노드 생성
    	x = (node *)malloc(sizeof(node));
    	x->key = k; x->l = x->r = tail;
    	// 부모랑 연결하고 split
    	if (k < p->key) p->l = x; else p->r = x;
    	split(k);
    }
    ```
